// Deklarace pole
int a[10];

// Zapis do pole
a[1] = 10;
a[3] = 99;

// Cteni z pole
write "a[1]: ", a[1];
write "10 + a[1]: ", 10 + a[1];
write "a[3]: ", a[3];
write "a[0] (default): ", a[0];

// Pole floatu
float f[5];
f[0] = 3.14;
f[2] = 2.71;
write "f[0]: ", f[0];
write "f[2]: ", f[2];

// Prace s polem v cyklu
int i;
int sum[1];
sum[0] = 0;
i = 0;
while (i < 5) {
    a[i] = i * i;
    sum[0] = sum[0] + a[i];
    i = i + 1;
}
write "a[0..4]: ", a[0], " ", a[1], " ", a[2], " ", a[3], " ", a[4];
write "suma ctvercu 0-4: ", sum[0];
